Clean Code Architecture (Laravel)

Implementing Service-Repository pattern for decoupled, testable logic.

SolarUsageService.php 

namespace App\Services;

class SolarUsageService implements UsageServiceInterface 
{
    public function __construct(
        private RepositoryInterface $repository,
        private NotificationSystem $notifier
    ) {}

    public function processDailyReport(int $userId): void 
    {
        $data = $this->repository->getByUser($userId);
        
        if ($data->isExceedingLimit()) {
            $this->notifier->sendAlert($userId, 'Usage Alert');
        }

        $this->repository->saveProcessedData($data);
    }
}
